include(FetchContent)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
add_executable(game-env main.cpp)
target_link_libraries(game-env PRIVATE birds_simulation ascent_physics nlohmann_json::nlohmann_json)
target_compile_options(game-env PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME environment-protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol_test.py $<TARGET_FILE:game-env>)
  set_tests_properties(environment-protocol PROPERTIES TIMEOUT 30)
endif()
