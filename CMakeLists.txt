cmake_minimum_required(VERSION 3.24)
project(tiny_rsi VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(GAMES_BUILD_UI "Build both native graphical games" ON)
include(CTest)
include(cmake/Sanitizers.cmake)
include(cmake/Dependencies.cmake)
add_subdirectory(games/angry-birds)
set(ASCENT_BUILD_GAME ${GAMES_BUILD_UI} CACHE BOOL "Build The Ascent UI" FORCE)
add_subdirectory(games/getting-over-it)
add_subdirectory(environments/native)
