cmake_minimum_required(VERSION 3.20)
project(TheAscent VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
option(ASCENT_BUILD_GAME "Build the native game" ON)
add_library(ascent_physics src/physics.cpp)
target_include_directories(ascent_physics PUBLIC src)
if(BUILD_TESTING)
 add_executable(physics_tests tests/physics_tests.cpp)
 target_link_libraries(physics_tests PRIVATE ascent_physics)
 add_test(NAME physics COMMAND physics_tests)
endif()
if(ASCENT_BUILD_GAME)
 find_package(raylib 5.5 QUIET)
 if(NOT raylib_FOUND)
  include(FetchContent)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(raylib)
 endif()
 add_executable(the_ascent src/main.cpp src/world.cpp)
 target_link_libraries(the_ascent PRIVATE ascent_physics raylib)
 target_compile_definitions(the_ascent PRIVATE ASCENT_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
 if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(the_ascent PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
  target_compile_options(ascent_physics PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
 endif()
 install(TARGETS the_ascent RUNTIME DESTINATION bin)
endif()
