cmake_minimum_required(VERSION 3.20)
project(TheAscent VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
option(ASCENT_BUILD_GAME "Build the native game" ON)
add_library(ascent_physics src/physics.cpp src/persistence.cpp)
target_include_directories(ascent_physics PUBLIC src)
if(BUILD_TESTING)
 add_executable(physics_tests tests/physics_tests.cpp)
 target_link_libraries(physics_tests PRIVATE ascent_physics)
 add_test(NAME physics COMMAND physics_tests)
 add_executable(persistence_tests tests/persistence_tests.cpp)
 target_link_libraries(persistence_tests PRIVATE ascent_physics)
 add_test(NAME persistence COMMAND persistence_tests)
endif()
if(ASCENT_BUILD_GAME)
 if(NOT TARGET raylib)
 find_package(raylib 5.5 QUIET)
 endif()
 if(NOT TARGET raylib)
  include(FetchContent)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz
   URL_HASH SHA256=aea98ecf5bc5c5e0b789a76de0083a21a70457050ea4cc2aec7566935f5e258e
   DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(raylib)
 endif()
 add_executable(the_ascent MACOSX_BUNDLE src/main.cpp src/world.cpp)
 set_target_properties(the_ascent PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/games/getting-over-it")
 set_target_properties(the_ascent PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "com.moksh.theascent" MACOSX_BUNDLE_BUNDLE_NAME "The Ascent" MACOSX_BUNDLE_SHORT_VERSION_STRING "1.0.0" MACOSX_BUNDLE_BUNDLE_VERSION "1" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/assets/Info.plist.in")
 if(APPLE)
  set(ASCENT_RESOURCE_DESTINATION "$<TARGET_FILE_DIR:the_ascent>/../Resources")
 else()
  set(ASCENT_RESOURCE_DESTINATION "$<TARGET_FILE_DIR:the_ascent>/assets")
 endif()
 add_custom_command(TARGET the_ascent POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "${ASCENT_RESOURCE_DESTINATION}")
 target_link_libraries(the_ascent PRIVATE ascent_physics raylib)
 target_compile_definitions(the_ascent PRIVATE ASCENT_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
 if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(the_ascent PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
  target_compile_options(ascent_physics PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
 endif()
 install(TARGETS the_ascent RUNTIME DESTINATION bin BUNDLE DESTINATION .)
 if(NOT APPLE)
  install(DIRECTORY assets/ DESTINATION share/the-ascent)
 endif()
endif()
